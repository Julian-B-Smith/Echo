const A = Math.pow(10, -6 / 20);
const sine = (f, a) => Array.from({ length: n }, (_, i) => a * Math.sin(2 * Math.PI * f * i / sr));
const chordHz = [55, 110, 130.81, 164.81];
const chord = () => { const y = new Array(n).fill(0); for (const f of chordHz) for (let i = 0; i < n; i++) y[i] += Math.sin(2 * Math.PI * f * i / sr); let pk = 0; for (const v of y) pk = Math.max(pk, Math.abs(v)); return y.map(v => v * A / pk); };
const cases = {};
{ const y = sine(1000, A); cases.sine_steady = y; }
{ const y = sine(1000, A).map((v, i) => i >= t0 ? v * 0.5 : v); cases.sine_step_half = y; }

for (const [k, y] of Object.entries(cases)) { const c = clickOf(Float64Array.from(y), sr, t0); console.log(`${k} ${c.db.toPrecision(17)} ${c.at.toPrecision(17)} ${c.floor.toPrecision(17)}`); }
